84// "<type> from_json: <what> "<input>"" with the input cut at 64 characters, built out of line from the hot path.
85[[noreturn]] inline void throw_json_error(std::string_view type, std::string_view what, std::string_view input)
86{
87 constexpr std::size_t max_quoted = 64;
88 std::string message(type);
89 message += " from_json: ";
90 message += what;
91 message += " \"";
92 message += input.substr(0, max_quoted);
93 if (input.size() > max_quoted) message += "...";
94 message += '"';
95 throw json_error(message);
96}

98// The string a text-form adapter reads, or a json_error naming the JSON type it got instead. Never throws nlohmann.
99inline const std::string& json_string_or_throw(const nlohmann::json& j, std::string_view type, std::string_view expected)
100{
101 const auto* s = j.get_ptr<const nlohmann::json::string_t*>();
102 if (s == nullptr) [[unlikely]] {
103 std::string what("expected ");
104 what += expected;
105 what += ", got";
106 throw_json_error(type, what, j.type_name());
107 }
108 return *s;
109}
