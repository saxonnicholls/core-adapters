107template <class... Args>
108[[nodiscard]] auto as_span(const ::nanobind::ndarray<Args...>& a) noexcept
109{
110 using array = ::nanobind::ndarray<Args...>;
111 static_assert(array::Order == 'C', "snicholls::python::as_span: the array must be C-contiguous (nanobind::c_contig)");
112 static_assert(array::DeviceType == ::nanobind::device::cpu::value,
113 "snicholls::python::as_span: the array must be on the CPU (nanobind::device::cpu)");
114 using scalar = typename array::Scalar;
115 return std::span<scalar>(a.data(), a.size());
116}

137template <class T>
138[[nodiscard]] numpy_matrix<T> to_numpy(std::vector<T>&& values, std::size_t rows, std::size_t columns)
139{
140 if (rows * columns != values.size()) [[unlikely]]
141 throw std::length_error("snicholls::python::to_numpy: rows * columns is not the number of values");
142 auto owned = std::make_unique<std::vector<T>>(std::move(values));
143 T* data = owned->data();
144 ::nanobind::capsule owner(owned.get(), &detail::delete_vector<T>);
145 owned.release();
146 return numpy_matrix<T>(data, {rows, columns}, owner);
147}

156template <class T>
157[[nodiscard]] numpy_vector<T> to_numpy(std::vector<T>&& values)
158{
159 const std::size_t n = values.size();
160 auto owned = std::make_unique<std::vector<T>>(std::move(values));
161 T* data = owned->data();
162 ::nanobind::capsule owner(owned.get(), &detail::delete_vector<T>);
163 owned.release();
164 return numpy_vector<T>(data, {n}, owner);
165}
