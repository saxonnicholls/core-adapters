# The snicholls_core Python package: the modules stats and game (nanobind, abi3 for Python 3.12+), the pure-Python
# files and the .pyi stubs from nanobind's stubgen. Everything is assembled in <build>/python/snicholls_core, an
# importable package; under scikit-build-core (SKBUILD) the modules and stubs are also installed into the wheel, and
# pyproject.toml's wheel.packages brings the pure-Python files.

set(_pkg ${CORE_ADAPTERS_PYTHON_DIR}/snicholls_core)
file(GLOB _pure CONFIGURE_DEPENDS ${CMAKE_CURRENT_SOURCE_DIR}/snicholls_core/*.py
     ${CMAKE_CURRENT_SOURCE_DIR}/snicholls_core/py.typed)
foreach(_f IN LISTS _pure)
    get_filename_component(_name ${_f} NAME)
    configure_file(${_f} ${_pkg}/${_name} COPYONLY)
endforeach()

# snicholls_core_add_module(<name>): python/src/<name>.cpp as snicholls_core.<name> (target snicholls_core_<name>)
function(snicholls_core_add_module name)
    # NOMINSIZE: nanobind optimises modules for size by default, which cost the numerics ~20% (BENCHMARKS.md)
    nanobind_add_module(snicholls_core_${name} STABLE_ABI NB_STATIC NOMINSIZE ${CMAKE_CURRENT_SOURCE_DIR}/src/${name}.cpp)
    target_link_libraries(snicholls_core_${name} PRIVATE snicholls::core_nanobind)
    set_target_properties(snicholls_core_${name} PROPERTIES OUTPUT_NAME ${name} LIBRARY_OUTPUT_DIRECTORY ${_pkg})
    core_adapters_compile_options(snicholls_core_${name})
    if(SKBUILD)
        install(TARGETS snicholls_core_${name} LIBRARY DESTINATION snicholls_core)
    endif()
endfunction()

snicholls_core_add_module(stats)
snicholls_core_add_module(game)

# Stubs: generated by importing the modules from the build tree, once both exist (game imports stats)
foreach(_m stats game)
    nanobind_add_stub(snicholls_core_${_m}_stub
        MODULE snicholls_core.${_m}
        OUTPUT ${_pkg}/${_m}.pyi
        PYTHON_PATH ${CORE_ADAPTERS_PYTHON_DIR}
        DEPENDS snicholls_core_stats snicholls_core_game)
    if(SKBUILD)
        install(FILES ${_pkg}/${_m}.pyi DESTINATION snicholls_core)
    endif()
endforeach()
