68template <utc_duration D>
69void to_json(nlohmann::json& j, const utc<D>& t)
70{
71 const rfc3339_text text = to_text(t);
72 if (!text) [[unlikely]]
73 throw json_error("snicholls::utc to_json: a year outside 0000..9999 has no RFC 3339 form");
74 j = std::string(text.view());

83template <utc_duration D>
84void from_json(const nlohmann::json& j, utc<D>& t)
85{
86 const std::string& s = detail::json_string_or_throw(j, "snicholls::utc", "an RFC 3339 string");
87 const parse_result<utc<D>> r = parse_utc<D>(s);
88 if (!r) [[unlikely]]
89 detail::throw_json_error("snicholls::utc",
90 r.error() == std::errc::result_out_of_range ? "out of range or too precise:"
91 : "not RFC 3339:",
92 s);
93 t = *r;
94}
