74// A held buffer (or a str's UTF-8), released with the GIL taken.
75class held_buffer
76{
77public:
78 held_buffer() noexcept = default;
79 held_buffer(const held_buffer&) = delete;
80 held_buffer& operator=(const held_buffer&) = delete;
81 held_buffer(held_buffer&& o) noexcept : buffer_(o.buffer_), held_(std::exchange(o.held_, false)),
82 text_(std::exchange(o.text_, nullptr)), data_(o.data_), size_(o.size_)
83 {
84 }
85 held_buffer& operator=(held_buffer&& o) noexcept
86 {
87 if (this != &o) {
88 release();
89 buffer_ = o.buffer_;
90 held_ = std::exchange(o.held_, false);
91 text_ = std::exchange(o.text_, nullptr);
92 data_ = o.data_;
93 size_ = o.size_;
94 }
95 return *this;
96 }
97 ~held_buffer() { release(); }
98
99 // A contiguous buffer; writable when asked. False, with no Python error left, when src has none.
100 bool acquire_buffer(PyObject* src, bool writable) noexcept
101 {
102 release();
103 if (PyUnicode_Check(src))
104 return false;
105 if (PyObject_GetBuffer(src, &buffer_, writable ? (PyBUF_SIMPLE | PyBUF_WRITABLE) : PyBUF_SIMPLE) != 0) {
106 PyErr_Clear();
107 return false;
108 }
109 held_ = true;
110 data_ = buffer_.buf;
111 size_ = static_cast<std::size_t>(buffer_.len);
112 return true;
113 }
114
115 // A str's UTF-8 (kept alive by a reference to the str), or a contiguous buffer.
116 bool acquire_text(PyObject* src) noexcept
117 {
118 release();
119 if (!PyUnicode_Check(src))
120 return acquire_buffer(src, false);
121 Py_ssize_t n = 0;
122 const char* s = PyUnicode_AsUTF8AndSize(src, &n);
123 if (s == nullptr) {
124 PyErr_Clear();
125 return false;
126 }
127 Py_INCREF(src);
128 text_ = src;
129 data_ = const_cast<char*>(s);
130 size_ = static_cast<std::size_t>(n);
131 return true;
132 }
133
134 [[nodiscard]] void* data() const noexcept { return data_; }
135 [[nodiscard]] std::size_t size() const noexcept { return size_; }
136
137private:
138 void release() noexcept
139 {
140 if (!held_ && text_ == nullptr)
141 return;
142 const PyGILState_STATE state = PyGILState_Ensure();
143 if (held_)
144 PyBuffer_Release(&buffer_);
145 Py_XDECREF(text_);
146 PyGILState_Release(state);
147 held_ = false;
148 text_ = nullptr;
149 data_ = nullptr;
150 size_ = 0;
151 }
152
153 Py_buffer buffer_{};
154 bool held_ = false;
155 PyObject* text_ = nullptr;
156 void* data_ = nullptr;
157 std::size_t size_ = 0;
158};
