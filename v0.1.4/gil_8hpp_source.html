169 R operator()(A... a) const
170 {
171 return with_gil([&]() -> R {
172 ::nanobind::object out = f_(std::forward<A>(a)...);
173 if constexpr (!std::is_void_v<R>) {
174 std::remove_cvref_t<R> value{};
175 if (!::nanobind::try_cast(out, value))
176 throw ::nanobind::type_error("snicholls::python::python_function: the callable's result does not "
177 "convert to the declared result type");
178 return value;
179 }
180 });
181 }

184 [[nodiscard]] const ::nanobind::object& object() const noexcept { return f_; }
